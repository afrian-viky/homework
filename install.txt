membuat db di phpmyadmin dengan nama db_sistemA

Tabel mahasiswa
CREATE TABLE mahasiswa (
  id INT AUTO_INCREMENT PRIMARY KEY,
  nama VARCHAR(100),
  nim VARCHAR(20),
  password VARCHAR(255)
);

Tabel buku
CREATE TABLE buku (
  id INT AUTO_INCREMENT PRIMARY KEY,
  judul VARCHAR(150),
  stok INT
);

Tabel peminjaman
CREATE TABLE peminjaman (
  id INT AUTO_INCREMENT PRIMARY KEY,
  mahasiswa_id INT,
  buku_id INT,
  status VARCHAR(20) DEFAULT 'pending'
);
